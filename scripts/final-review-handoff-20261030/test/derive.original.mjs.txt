import test from 'node:test';import assert from 'node:assert/strict';import fs from 'node:fs';
import {deriveFinalReview,matchesFinalReview,VERSION} from '../derive.mjs';
function material(){return {context:{storeId:'FictionalStore',month:'2026-11',timezone:'America/New_York',principalEpoch:1},result:JSON.parse(fs.readFileSync('/workspace/sage-complete-fictional-pair/proposal.json','utf8'))};}
const derive=m=>deriveFinalReview(m,{enabled:true});
test('default disabled does not inspect material',()=>{const m={get result(){throw Error('must not read');}};assert.equal(deriveFinalReview(m).state,'disabled');});
test('full actual manager proposal yields 28 explicit off items and typed unknowns without mutation or approval',()=>{const m=material(),before=structuredClone(m),d=derive(m);assert.equal(d.version,VERSION);assert.equal(d.proposedOff.length,28);assert.equal(d.projection.length,2);assert.equal(d.projection[0].days[0].status,'unknown');assert.equal(d.trusted,false);assert.equal(d.publicationAuthorized,false);assert.ok(d.issues.some(x=>x.code==='unknown-membership'));assert.deepEqual(m,before);assert.ok(Object.isFrozen(d.preserved.result.proposal));assert.ok(matchesFinalReview(d,m));for(const x of d.proposedOff){assert.equal(x.kind,'off');assert.equal(x.absence.source.kind,'generated preview');assert.ok(d.projection.find(p=>p.personId===x.personId).days.some(day=>day.date===x.date&&day.status==='off'));}});
test('missing or null kind/provenance never falls back to off',()=>{for(const absence of [{source:{kind:'generated preview'}},{kind:null,source:{kind:'generated preview'}},{kind:'off'},null]){const m=material();m.result.proposal.records['2026-11-02'].absences.FictionalB=absence;const d=derive(m);assert.equal(d.proposedOff.some(x=>x.date==='2026-11-02'&&x.personId==='FictionalB'),false);assert.ok(d.issues.some(x=>x.code==='explicit-absence-kind-and-provenance-required'));assert.deepEqual(d.preserved.result.proposal.records['2026-11-02'].absences.FictionalB,absence);}});
test('vacation unavailable and unknown remain typed and never join proposed off',()=>{for(const kind of ['vacation','unavailable','unknown']){const m=material();m.result.proposal.records['2026-11-02'].absences.FictionalB.kind=kind;const d=derive(m);assert.equal(d.projection[1].days.find(x=>x.date==='2026-11-02').status,kind);assert.equal(d.proposedOff.some(x=>x.personId==='FictionalB'&&x.date==='2026-11-02'),false);}});
test('split shifts and fractional/negative-zero values remain exact with temporal blockers',()=>{const m=material(),r=m.result.proposal.records['2026-11-02'],s=r.shifts[0];r.shifts=[{...s,startMinute:-0,endMinute:720.5},{...s,startMinute:780,endMinute:1020}];const d=derive(m);const day=d.projection[0].days.find(x=>x.date==='2026-11-02');assert.deepEqual(day.intervals,[[-0,720.5],[780,1020]]);assert.ok(Object.is(day.intervals[0][0],-0));assert.ok(d.issues.some(x=>x.code==='temporal-publication-unsupported'));assert.deepEqual(d.preserved.result.proposal.records['2026-11-02'].shifts,r.shifts);});
test('source, principal, interval, absence and diagnostics changes stale exact material',()=>{const original=material(),d=derive(original);for(const mutate of [m=>m.context.principalEpoch++,m=>m.result.proposal.records['2026-11-02'].absences.FictionalB.source.kind='edited',m=>m.result.proposal.records['2026-11-02'].shifts[0].endMinute--,m=>m.result.proposal.days[0].notes.push('changed')]){const next=structuredClone(original);mutate(next);assert.equal(matchesFinalReview(d,next),false);}});
test('case identity mismatch refuses; conflicting assignments preserved but not selected',()=>{const m=material();m.context.storeId='fictionalstore';assert.throws(()=>derive(m),/scoped/);const n=material();n.result.proposal.records['2026-11-02'].on.push('FictionalB');const d=derive(n);assert.ok(d.issues.some(x=>x.code==='conflicting-state'));assert.equal(d.proposedOff.some(x=>x.date==='2026-11-02'&&x.personId==='FictionalB'),false);assert.ok(d.preserved.result.proposal.records['2026-11-02'].on.includes('FictionalB'));});
test('missing dates remain missing and incomplete, not invented off or work',()=>{const m=material();delete m.result.proposal.records['2026-11-02'];const d=derive(m);assert.ok(d.issues.some(x=>x.code==='record-missing'));assert.ok(d.projection.every(p=>!p.days.some(x=>x.date==='2026-11-02')));});
test('accessors refused without execution, undefined retained and distinct from missing',()=>{let called=false;assert.throws(()=>derive({get result(){called=true;return {};}}));assert.equal(called,false);const m=material();m.result.proposal.extra=undefined;const d=derive(m);assert.ok(Object.hasOwn(d.preserved.result.proposal,'extra'));const n=structuredClone(m);delete n.result.proposal.extra;assert.equal(matchesFinalReview(d,n),false);});
test('explicit closed unknown and open-unentered dates remain distinct without invented off',()=>{for(const [state,status] of [['closed','closed'],['unknown','unknown'],['open','unentered']]){const m=material();m.result.proposal.records['2026-11-02']={on:[],off:[],shifts:[]};m.result.proposal.days.find(d=>d.day==='2026-11-02').state=state;const d=derive(m);for(const person of d.projection){const day=person.days.find(x=>x.date==='2026-11-02');assert.equal(day.status,status);assert.deepEqual(day.intervals,[]);}assert.equal(d.proposedOff.some(x=>x.date==='2026-11-02'),false);}});
